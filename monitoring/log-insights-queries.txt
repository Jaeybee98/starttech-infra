# Query 1: Filter application panic attacks and runtime errors
fields @timestamp, @message, @logStream
| filter @message like /ERROR/ or @message like /panic/ or @message like /failure/
| sort @timestamp desc
| limit 50

# Query 2: HTTP Request Performance Analysis
fields @timestamp, @message
| filter @message like /HTTP/
| parse @message "* * * *" as method, path, status, latency
| stats count() by status, path
| sort count() desc
